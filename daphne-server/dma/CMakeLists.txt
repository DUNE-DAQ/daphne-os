cmake_minimum_required(VERSION 3.10)
project(daphne_dma_diagnostics LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "DMA diagnostics require Linux; use WSL or native DAPHNE")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_library(daphne_afe_dma afe_capture.cpp afe_trigger_capture.cpp)
target_include_directories(daphne_afe_dma PUBLIC .)
target_compile_options(daphne_afe_dma PRIVATE -Wall -Wextra -Wpedantic)
add_library(daphne_dma capture.cpp)
target_include_directories(daphne_dma PUBLIC . ../kernel/daphne-dma-diag)
target_compile_options(daphne_dma PRIVATE -Wall -Wextra -Wpedantic)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_library(daphne_dma_linux linux_backend.cpp)
  target_link_libraries(daphne_dma_linux PUBLIC daphne_dma)
  target_compile_options(daphne_dma_linux PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(daphne-dma-diag main.cpp)
  target_link_libraries(daphne-dma-diag PRIVATE daphne_dma_linux)
  target_compile_options(daphne-dma-diag PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(afe_capture_tests afe_capture_tests.cpp)
  target_link_libraries(afe_capture_tests PRIVATE daphne_afe_dma)
  add_test(NAME afe_capture_unit COMMAND afe_capture_tests)
  add_executable(afe_trigger_tests afe_trigger_tests.cpp)
  target_link_libraries(afe_trigger_tests PRIVATE daphne_afe_dma)
  add_test(NAME afe_trigger_unit COMMAND afe_trigger_tests)
  add_executable(dma_capture_tests tests.cpp)
  target_link_libraries(dma_capture_tests PRIVATE daphne_dma)
  target_compile_options(dma_capture_tests PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME dma_capture_unit COMMAND dma_capture_tests)
  set_tests_properties(dma_capture_unit PROPERTIES LABELS unit)
  if(TARGET daphne-dma-diag)
    add_test(NAME dma_selection_omitted COMMAND daphne-dma-diag)
    set_tests_properties(dma_selection_omitted PROPERTIES
      LABELS unit PASS_REGULAR_EXPRESSION "spybuffer.*DMA not initialized")
  endif()
endif()

if(BUILD_TESTING)
  add_executable(afe_eye_tests afe_eye_tests.cpp)
  target_link_libraries(afe_eye_tests PRIVATE daphne_afe_dma)
  add_test(NAME afe_eye_unit COMMAND afe_eye_tests)
endif()
